// ANTIPATRÓN C++: Asumir que la GPU tiene cachés mágicas que absorben lecturas redundantes
@group(0) @binding(0) var<storage, read> inData: array<f32>;
@group(0) @binding(1) var<storage, read_write> outData: array<f32>;

const RADIUS: i32 = 16;

@compute @workgroup_size(64)
fn main(@builtin(global_invocation_id) global_id: vec3<u32>) {
    let i = i32(global_id.x);
    let n = i32(arrayLength(&inData));
    if (i >= n) { return; }

    var sum: f32 = 0.0;

    // ❌ ERROR DE CONCEPTO: Cada uno de los 64 hilos lee 33 veces de VRAM global
    // 64 hilos * 33 lecturas = 2.112 accesos a memoria lenta (~800 ciclos de latencia)
    for (var k: i32 = -RADIUS; k <= RADIUS; k = k + 1) {
        let sample_idx = clamp(i + k, 0, n - 1);
        sum = sum + inData[u32(sample_idx)];
    }

    outData[global_id.x] = sum / f32(2 * RADIUS + 1);
}
// WGSL IDIOMÁTICO: Memoria compartida en chip 'var<workgroup>' y carga cooperativa
@group(0) @binding(0) var<storage, read> inData: array<f32>;
@group(0) @binding(1) var<storage, read_write> outData: array<f32>;

const RADIUS: u32 = 16u;
const WORKGROUP_SIZE: u32 = 64u;
const TILE_SIZE: u32 = 96u; // 64 + 2 * 16

// ✅ MEMORIA COMPARTIDA EN CHIP (SRAM ultrarrápida, ~20 ciclos, común a todo el workgroup)
var<workgroup> tile: array<f32, 96>;

@compute @workgroup_size(64)
fn main(
    @builtin(global_invocation_id) global_id: vec3<u32>,
    @builtin(local_invocation_id) local_id: vec3<u32>,
    @builtin(workgroup_id) group_id: vec3<u32>
) {
    let g_idx = global_id.x;
    let l_idx = local_id.x;
    let n = arrayLength(&inData);
    let base_idx = group_id.x * WORKGROUP_SIZE;

    // 1. CARGA COOPERATIVA: Los 64 hilos colaboran para rellenar la baldosa en SRAM
    // A) Cada uno de los 64 hilos carga su valor central
    let center_g = min(base_idx + l_idx, n - 1u);
    tile[l_idx + RADIUS] = inData[center_g];

    // B) Los primeros 16 hilos (del 0 al 15) cargan el halo izquierdo
    if (l_idx < RADIUS) {
        let left_g = i32(base_idx) - i32(RADIUS) + i32(l_idx);
        tile[l_idx] = inData[u32(max(left_g, 0))];
    }

    // C) Los siguientes 16 hilos (del 16 al 31) cargan el halo derecho
    if (l_idx >= RADIUS && l_idx < 2u * RADIUS) {
        let offset = l_idx - RADIUS;
        let right_g = min(base_idx + WORKGROUP_SIZE + offset, n - 1u);
        tile[WORKGROUP_SIZE + RADIUS + offset] = inData[right_g];
    }

    // 2. BARRERA HARDWARE: Todos los 96 valores ya están garantizados en SRAM
    workgroupBarrier();

    if (g_idx >= n) { return; }

    // 3. CÓMPUTO A VELOCIDAD DE REGISTRO / SRAM (Cero accesos a VRAM lenta)
    var sum: f32 = 0.0;
    for (var k: u32 = 0u; k <= 2u * RADIUS; k = k + 1u) {
        sum = sum + tile[l_idx + k];
    }

    outData[g_idx] = sum / f32(2u * RADIUS + 1u);
}

1. El «Muro de la Memoria» (The Memory Wall):
Como demuestran David A. Patterson y John L. Hennessy en su obra canónica Computer Architecture: A Quantitative Approach, la velocidad de procesamiento de las ALUs en las GPUs supera en varios órdenes de magnitud el ancho de banda del bus de memoria exterior (DRAM/VRAM). Mientras que la VRAM global requiere cientos de ciclos de latencia (~600–800 ciclos), la memoria compartida integrada en cada núcleo (SRAM) entrega datos en solo ~20 ciclos con un ancho de banda interno que supera 1,5 TB/s.

2. La falacia de la caché invisible de CPU en silicio SIMT:
En C++ sobre CPU, el programador confía ciegamente en que las cachés transparentes L1/L2 mitigarán los accesos repetidos. En una GPU con miles de hilos activos simultáneos, esa aproximación destruye el rendimiento: las líneas de caché son expulsadas continuamente (*cache thrashing*), saturando el bus de memoria.

3. La semántica de WGSL como lenguaje de GPU:
WGSL exige declarar explícitamente el espacio de direcciones con var<workgroup> y sincronizar con workgroupBarrier(). No es una limitación sintáctica, sino la herramienta que permite a los 64 hilos cooperar para descargar el 95,5% del tráfico de memoria externa, logrando una aceleración real, medible y físicamente garantizada en cualquier arquitectura (Apple Silicon, NVIDIA o AMD).

4. ¿Por qué la diferencia es moderada y por qué es crucial en producción?
Al comparar CPU frente a GPU vimos saltos de 15x a 30x porque pasamos de unos pocos núcleos a miles de hilos paralelos. En este taller, en cambio, ambos códigos se ejecutan dentro de la GPU, por lo que la diferencia no es un salto de órdenes de magnitud. Sin embargo, en una aplicación real (como procesar vídeo a 60 FPS, un motor gráfico o inferencia de redes neuronales), esta operación se repite millones de veces de forma continuada. Multiplicada en cada fotograma, esta optimización marca la frontera entre un rendimiento fluido a 60/120 FPS o el estrangulamiento térmico y el drenaje de batería.
