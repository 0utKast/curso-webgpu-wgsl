Convolución Optimizada mediante Región de Halo en SRAM

En lugar de que cada hilo emita 9 lecturas independientes a la memoria externa, los 256 hilos cooperan para cargar una baldosa ampliada de 18×18 elementos en la memoria compartida on-chip. A partir de ese instante, todas las operaciones de convolución se resuelven a la velocidad del registro físico (1-2 ciclos).

"La única forma de romper este muro y catapultar la velocidad es cooperar: cargar los datos una sola vez en la memoria compartida on-chip..."

1// Recursos: Textura de entrada (muestreable) y salida (almacenamiento)
2@group(0) @binding(0) var textura_entrada: texture_2d<f32>;
3@group(0) @binding(1) var textura_salida: texture_storage_2d<rgba8unorm, write>;
4
5// Dimensiones del Workgroup y la Región de Halo (R = 1)
6const TILE_ANCHO: u32 = 16u;
7const TILE_ALTO: u32 = 16u;
8const HALO_ANCHO: u32 = 18u; // 16 + 2 * 1
9const TOTAL_PIXELES_HALO: u32 = 324u; // 18 x 18 = 324 píxeles
10const TOTAL_HILOS_WORKGROUP: u32 = 256u; // 16 x 16 = 256 hilos
11
12// Baldosa compartida en SRAM on-chip: 324 elementos f32 = 1.296 bytes (1.3 KB)
13var<workgroup> baldosa_halo: array<array<f32, 18>, 18>;
14
15// Función auxiliar: Clamping de coordenadas y luminancia perceptual
16fn leer_luminancia_global(coords: vec2<i32>, dims: vec2<i32>) -> f32 {
17 let coords_seguras = vec2<u32>(
18 clamp(coords, vec2<i32>(0, 0), dims - vec2<i32>(1, 1))
19 );
20 let color: vec4<f32> = textureLoad(textura_entrada, coords_seguras, 0u);
21 return 0.299 * color.r + 0.587 * color.g + 0.114 * color.b;
22}
23
24@compute @workgroup_size(16, 16)
25fn cs_sobel_halo(
26 @builtin(global_invocation_id) global_id: vec3<u32>,
27 @builtin(local_invocation_id) local_id: vec3<u32>,
28 @builtin(workgroup_id) group_id: vec3<u32>
29) {
30 let dims = vec2<i32>(textureDimensions(textura_entrada));
31
32 // Origen global del bloque en la imagen (esquina superior izquierda interior)
33 let base_bloque: vec2<i32> = vec2<i32>(group_id.xy) * vec2<i32>(16, 16);
34
35 // Índice lineal del hilo dentro del bloque (0 a 255)
36 let hilo_lineal: u32 = local_id.y * 16u + local_id.x;
37
38 // 3. CARGA COOPERATIVA EN SRAM (Strided Loop):
39 // 256 hilos cargan 324 píxeles (los primeros 68 hilos cargan 2 píxeles)
40 for (var i: u32 = hilo_lineal; i < TOTAL_PIXELES_HALO; i += TOTAL_HILOS_WORKGROUP) {
41 let halo_y: u32 = i / HALO_ANCHO; // Fila local (0 a 17)
42 let halo_x: u32 = i % HALO_ANCHO; // Columna local (0 a 17)
43
44 // Coordenada global en la imagen con offset -1 de halo
45 let coords_globales = base_bloque + vec2<i32>(i32(halo_x) - 1, i32(halo_y) - 1);
46 baldosa_halo[halo_y][halo_x] = leer_luminancia_global(coords_globales, dims);
47 }
48
49 // 4. BARRERA DE SINCRONIZACIÓN: Asegura que la baldosa SRAM está llena
50 workgroupBarrier();
51
52 // 5. Guarda de límites para descartar píxeles sobrantes de escritura
53 if (i32(global_id.x) >= dims.x || i32(global_id.y) >= dims.y) {
54 return;
55 }
56
57 // 6. LECTURA 3x3 DIRECTA DESDE SRAM ON-CHIP (1-2 ciclos de latencia)
58 let lx: u32 = local_id.x + 1u;
59 let ly: u32 = local_id.y + 1u;
60
61 let l00 = baldosa_halo[ly - 1u][lx - 1u];
62 let l10 = baldosa_halo[ly - 1u][lx];
63 let l20 = baldosa_halo[ly - 1u][lx + 1u];
64 let l01 = baldosa_halo[ly][lx - 1u];
65 let l21 = baldosa_halo[ly][lx + 1u];
66 let l02 = baldosa_halo[ly + 1u][lx - 1u];
67 let l12 = baldosa_halo[ly + 1u][lx];
68 let l22 = baldosa_halo[ly + 1u][lx + 1u];
69
70 // 7. Convolución Sobel (Kx y Ky) y Magnitud Euclídea
71 let gx = (-1.0 * l00) + (1.0 * l20) +
72 (-2.0 * l01) + (2.0 * l21) +
73 (-1.0 * l02) + (1.0 * l22);
74
75 let gy = (-1.0 * l00) + (-2.0 * l10) + (-1.0 * l20) +
76 ( 1.0 * l02) + ( 2.0 * l12) + ( 1.0 * l22);
77
78 let magnitud = sqrt(gx * gx + gy * gy);
79
80 // Escritura directa en VRAM
81 textureStore(textura_salida, vec2<u32>(global_id.xy), vec4<f32>(magnitud, magnitud, magnitud, 1.0));
82}
